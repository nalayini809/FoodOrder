CREATE TABLE FOODORDER_TB (
        ORDERID VARCHAR2(12 BYTE) PRIMARY KEY,
        CUSTOMERNAME VARCHAR2(50 BYTE) NOT NULL,
        FOODITEM VARCHAR2(50 BYTE) NOT NULL,
        ORDER_DATE DATE NOT NULL,
        QUANTITY NUMBER,
        PRICE NUMBER,
        REMARKS VARCHAR2(100 BYTE)
    );
    
CREATE SEQUENCE FOODORDER_SEQ  START WITH 10 INCREMENT BY 1 MAXVALUE 99;

SELECT * FROM FOODORDER_TB;
